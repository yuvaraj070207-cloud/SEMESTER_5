#include "ns3/core-module.h"
#include "ns3/network-module.h"
#include "ns3/internet-module.h"
#include "ns3/point-to-point-module.h"
#include "ns3/applications-module.h"
#include "ns3/flow-monitor-module.h"
#include <cmath>

using namespace ns3;

NS_LOG_COMPONENT_DEFINE("TcpComparison");

static void
CwndChange(Ptr<OutputStreamWrapper> stream, uint32_t oldCwnd, uint32_t newCwnd)
{
    *stream->GetStream() << Simulator::Now().GetSeconds() << "\t" << newCwnd << std::endl;
}

int main(int argc, char *argv[])
{
    double errorRate = 0.001; // Packet Error Rate
    CommandLine cmd(__FILE__);
    cmd.AddValue("errorRate", "Packet loss rate on bottleneck link", errorRate);
    cmd.Parse(argc, argv);

    // Create Topology: 2 Senders (n0, n1) -> 2 Routers (r0, r1) -> 2 Receivers (n2, n3)
    NodeContainer senders, receivers, routers;
    senders.Create(2);
    receivers.Create(2);
    routers.Create(2);

    // Access Links (Senders to Router 0)
    PointToPointHelper p2pAccess;
    p2pAccess.SetDeviceAttribute("DataRate", StringValue("100Mbps"));
    p2pAccess.SetChannelAttribute("Delay", StringValue("2ms"));

    NetDeviceContainer dev0 = p2pAccess.Install(senders.Get(0), routers.Get(0));
    NetDeviceContainer dev1 = p2pAccess.Install(senders.Get(1), routers.Get(0));

    // Bottleneck Link (Router 0 to Router 1)
    PointToPointHelper p2pBottleneck;
    p2pBottleneck.SetDeviceAttribute("DataRate", StringValue("10Mbps"));
    p2pBottleneck.SetChannelAttribute("Delay", StringValue("20ms"));
    p2pBottleneck.SetQueue("ns3::DropTailQueue<Packet>", "MaxSize", StringValue("30p"));

    // Add Error Model to bottleneck to simulate loss
    Ptr<RateErrorModel> em = CreateObject<RateErrorModel>();
    em->SetAttribute("ErrorRate", DoubleValue(errorRate));
    em->SetAttribute("ErrorUnit", StringValue("ERROR_UNIT_PACKET"));

    NetDeviceContainer devBottleneck = p2pBottleneck.Install(routers.Get(0), routers.Get(1));
    devBottleneck.Get(1)->SetAttribute("ReceiveErrorModel", PointerValue(em));

    // Egress Links (Router 1 to Receivers)
    NetDeviceContainer dev2 = p2pAccess.Install(routers.Get(1), receivers.Get(2));
    NetDeviceContainer dev3 = p2pAccess.Install(routers.Get(1), receivers.Get(3));

    // Assign dynamic TCP Type per node
    InternetStackHelper stackNewReno, stackCubic;
    
    // Node 0: TCP NewReno
    Config::SetDefault("ns3::TcpL4Protocol::SocketType", StringValue("ns3::TcpNewReno"));
    stackNewReno.Install(senders.Get(0));
    stackNewReno.Install(receivers.Get(0));

    // Node 1: TCP Cubic
    Config::SetDefault("ns3::TcpL4Protocol::SocketType", StringValue("ns3::TcpCubic"));
    stackCubic.Install(senders.Get(1));
    stackCubic.Install(receivers.Get(1));

    // Router Stack
    InternetStackHelper stackRouter;
    stackRouter.Install(routers);

    // IP Addressing
    Ipv4AddressHelper address;
    
    address.SetBase("10.1.1.0", "255.255.255.0");
    Ipv4InterfaceContainer ip0 = address.Assign(dev0);

    address.SetBase("10.1.2.0", "255.255.255.0");
    Ipv4InterfaceContainer ip1 = address.Assign(dev1);

    address.SetBase("10.1.3.0", "255.255.255.0");
    Ipv4InterfaceContainer ipB = address.Assign(devBottleneck);

    address.SetBase("10.1.4.0", "255.255.255.0");
    Ipv4InterfaceContainer ip2 = address.Assign(dev2);

    address.SetBase("10.1.5.0", "255.255.255.0");
    Ipv4InterfaceContainer ip3 = address.Assign(dev3);

    Ipv4GlobalRoutingHelper::PopulateRoutingTables();

    // Applications: Flow 1 (NewReno)
    uint16_t port1 = 9001;
    BulkSendHelper send1("ns3::TcpSocketFactory", InetSocketAddress(ip2.GetAddress(1), port1));
    ApplicationContainer app1 = send1.Install(senders.Get(0));
    app1.Start(Seconds(1.0));
    app1.Stop(Seconds(40.0));

    PacketSinkHelper sink1("ns3::TcpSocketFactory", InetSocketAddress(Ipv4Address::GetAny(), port1));
    ApplicationContainer sinkApp1 = sink1.Install(receivers.Get(0));
    sinkApp1.Start(Seconds(0.0));
    sinkApp1.Stop(Seconds(40.0));

    // Applications: Flow 2 (Cubic)
    uint16_t port2 = 9002;
    BulkSendHelper send2("ns3::TcpSocketFactory", InetSocketAddress(ip3.GetAddress(1), port2));
    ApplicationContainer app2 = send2.Install(senders.Get(1));
    app2.Start(Seconds(1.0));
    app2.Stop(Seconds(40.0));

    PacketSinkHelper sink2("ns3::TcpSocketFactory", InetSocketAddress(Ipv4Address::GetAny(), port2));
    ApplicationContainer sinkApp2 = sink2.Install(receivers.Get(1));
    sinkApp2.Start(Seconds(0.0));
    sinkApp2.Stop(Seconds(40.0));

    // Tracing CWND
    AsciiTraceHelper ascii;
    Ptr<OutputStreamWrapper> stream1 = ascii.CreateFileStream("cwnd-newreno.txt");
    Ptr<OutputStreamWrapper> stream2 = ascii.CreateFileStream("cwnd-cubic.txt");

    Simulator::Schedule(Seconds(1.001), [] (Ptr<OutputStreamWrapper> st) {
        Config::ConnectWithoutContext("/NodeList/0/$ns3::TcpL4Protocol/SocketList/0/CongestionWindow", MakeBoundCallback(&CwndChange, st));
    }, stream1);

    Simulator::Schedule(Seconds(1.001), [] (Ptr<OutputStreamWrapper> st) {
        Config::ConnectWithoutContext("/NodeList/1/$ns3::TcpL4Protocol/SocketList/0/CongestionWindow", MakeBoundCallback(&CwndChange, st));
    }, stream2);

    // Metrics Collection via FlowMonitor
    FlowMonitorHelper flowmon;
    Ptr<FlowMonitor> monitor = flowmon.InstallAll();

    Simulator::Stop(Seconds(40.0));
    Simulator::Run();

    monitor->CheckForLostPackets();
    FlowMonitor::FlowStatsContainer stats = monitor->GetFlowStats();

    double t1 = 0, t2 = 0;
    for (auto iter = stats.begin(); iter != stats.end(); ++iter)
    {
        if (iter->first == 1) t1 = (iter->second.rxBytes * 8.0) / (39.0 * 1e6);
        if (iter->first == 2) t2 = (iter->second.rxBytes * 8.0) / (39.0 * 1e6);
    }

    // Jain's Fairness Index Calculation: J = (x1 + x2)^2 / (2 * (x1^2 + x2^2))
    double jainIndex = (std::pow(t1 + t2, 2)) / (2.0 * (std::pow(t1, 2) + std::pow(t2, 2)));

    std::cout << "\n=== COMPARATIVE ANALYSIS ===" << std::endl;
    std::cout << "TCP NewReno Throughput: " << t1 << " Mbps" << std::endl;
    std::cout << "TCP Cubic Throughput:   " << t2 << " Mbps" << std::endl;
    std::cout << "Jain's Fairness Index:  " << jainIndex << std::endl;

    Simulator::Destroy();
    return 0;
}
