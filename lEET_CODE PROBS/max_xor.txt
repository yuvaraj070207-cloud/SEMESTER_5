int findMaximumXOR(int* nums, int numsSize)
{
    int n=numsSize;
    int maxRes=0;
    int temp;
    for(int i=0;i<n-1;i++)
    {
        for(int j=i+1;j<n;j++)
        {
        
            temp=nums[i]^nums[j];
            if(temp>maxRes)
            {
                maxRes=temp;
            }

        }
    }
    return maxRes;

}
