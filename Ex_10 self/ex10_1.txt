#include "ns3/core-module.h"
#include "ns3/network-module.h"
#include "ns3/internet-module.h"
#include "ns3/point-to-point-module.h"
#include "ns3/applications-module.h"
#include "ns3/flow-monitor-module.h"

using namespace ns3;

NS_LOG_COMPONENT_DEFINE("TcpSingleFlow");

// Callback to trace Congestion Window (cwnd) changes
static void
CwndChange(Ptr<OutputStreamWrapper> stream, uint32_t oldCwnd, uint32_t newCwnd)
{
    *stream->GetStream() << Simulator::Now().GetSeconds() << "\t" << newCwnd << std::endl;
}

int main(int argc, char *argv[])
{
    // Set default TCP Congestion Control Algorithm (e.g., TcpNewReno)
    Config::SetDefault("ns3::TcpL4Protocol::SocketType", StringValue("ns3::TcpNewReno"));
    
    // Create 3 nodes: Sender (n0) -> Router (n1) -> Receiver (n2)
    NodeContainer nodes;
    nodes.Create(3);

    // Node 0 to Node 1 link (High bandwidth, low delay)
    PointToPointHelper p2pSenderRouter;
    p2pSenderRouter.SetDeviceAttribute("DataRate", StringValue("100Mbps"));
    p2pSenderRouter.SetChannelAttribute("Delay", StringValue("2ms"));

    NetDeviceContainer d0d1 = p2pSenderRouter.Install(nodes.Get(0), nodes.Get(1));

    // Node 1 to Node 2 link (Bottleneck link)
    PointToPointHelper p2pRouterReceiver;
    p2pRouterReceiver.SetDeviceAttribute("DataRate", StringValue("10Mbps"));
    p2pRouterReceiver.SetChannelAttribute("Delay", StringValue("10ms"));
    p2pRouterReceiver.SetQueue("ns3::DropTailQueue<Packet>", "MaxSize", StringValue("20p"));

    NetDeviceContainer d1d2 = p2pRouterReceiver.Install(nodes.Get(1), nodes.Get(2));

    // Install Internet Stack
    InternetStackHelper stack;
    stack.Install(nodes);

    // Assign IP Addresses
    Ipv4AddressHelper address;
    address.SetBase("10.1.1.0", "255.255.255.0");
    Ipv4InterfaceContainer i0i1 = address.Assign(d0d1);

    address.SetBase("10.1.2.0", "255.255.255.0");
    Ipv4InterfaceContainer i1i2 = address.Assign(d1d2);

    Ipv4GlobalRoutingHelper::PopulateRoutingTables();

    // Setup BulkSend Application on Sender (Node 0)
    uint16_t port = 9;
    BulkSendHelper source("ns3::TcpSocketFactory", InetSocketAddress(i1i2.GetAddress(1), port));
    source.SetAttribute("MaxBytes", UintegerValue(0)); // Unlimited data
    ApplicationContainer sourceApps = source.Install(nodes.Get(0));
    sourceApps.Start(Seconds(1.0));
    sourceApps.Stop(Seconds(35.0));

    // Setup PacketSink Application on Receiver (Node 2)
    PacketSinkHelper sink("ns3::TcpSocketFactory", InetSocketAddress(Ipv4Address::GetAny(), port));
    ApplicationContainer sinkApps = sink.Install(nodes.Get(2));
    sinkApps.Start(Seconds(0.0));
    sinkApps.Stop(Seconds(35.0));

    // Trace Congestion Window (cwnd)
    AsciiTraceHelper asciiTraceHelper;
    Ptr<OutputStreamWrapper> stream = asciiTraceHelper.CreateFileStream("cwnd-single.txt");
    Simulator::Schedule(Seconds(1.001), [] (Ptr<OutputStreamWrapper> st) {
        Config::ConnectWithoutContext("/NodeList/0/$ns3::TcpL4Protocol/SocketList/0/CongestionWindow", MakeBoundCallback(&CwndChange, st));
    }, stream);

    // FlowMonitor for metrics calculation
    FlowMonitorHelper flowmon;
    Ptr<FlowMonitor> monitor = flowmon.InstallAll();

    Simulator::Stop(Seconds(35.0));
    Simulator::Run();

    // Print Performance Metrics
    monitor->CheckForLostPackets();
    Ptr<Ipv4FlowClassifier> classifier = DynamicCast<Ipv4FlowClassifier>(flowmon.GetClassifier());
    FlowMonitor::FlowStatsContainer stats = monitor->GetFlowStats();

    for (auto iter = stats.begin(); iter != stats.end(); ++iter)
    {
        Ipv4FlowProbeTag ptag;
        FlowId flowId = iter->first;
        if (flowId == 1) // Sender flow
        {
            double duration = iter->second.timeLastRxPacket.GetSeconds() - iter->second.timeFirstTxPacket.GetSeconds();
            double throughput = (iter->second.rxBytes * 8.0) / (duration * 1e6); // Mbps
            double pdr = (double)iter->second.rxPackets / iter->second.txPackets * 100.0;
            double avgDelay = iter->second.delaySum.GetSeconds() / iter->second.rxPackets;

            std::cout << "--- PERFORMANCE METRICS ---" << std::endl;
            std::cout << "Throughput: " << throughput << " Mbps" << std::endl;
            std::cout << "Packet Delivery Ratio (PDR): " << pdr << " %" << std::endl;
            std::cout << "Packet Loss: " << iter->second.lostPackets << " packets" << std::endl;
            std::cout << "Average Delay: " << avgDelay * 1000 << " ms" << std::endl;
        }
    }

    Simulator::Destroy();
    return 0;
}
